function showTab(name){
  document.getElementById('quiz').classList.toggle('active', name==='quiz');
  document.getElementById('flash').classList.toggle('active', name==='flash');
  document.getElementById('tab-quiz').classList.toggle('active', name==='quiz');

/* ---------------- QUIZ LOGIC ---------------- */
const userAnswers = new Array(quizData.length).fill(null);
let corrected = false;

function renderQuiz(){
  const container = document.getElementById('quiz-container');
  container.innerHTML = '';
  quizData.forEach((q, qi) => {
    const card = document.createElement('div');
    card.className = 'question-card';
    card.id = 'q-card-' + qi;

    const tag = document.createElement('div');
    tag.className = 'question-tag';
    tag.textContent = q.tag;
    card.appendChild(tag);

    const prompt = document.createElement('p');
    prompt.className = 'question-text';
    prompt.textContent = (qi+1) + '. ' + q.prompt;
    card.appendChild(prompt);

    const opts = document.createElement('div');
    opts.className = 'options';
    q.options.forEach((optText, oi) => {
      const label = document.createElement('label');
      label.className = 'option';
      label.id = 'opt-' + qi + '-' + oi;
      const input = document.createElement('input');
      input.type = 'radio';
      input.name = 'q' + qi;
      input.value = oi;
      input.onchange = () => { userAnswers[qi] = oi; clearError(qi); };
      label.appendChild(input);
      const span = document.createElement('span');
      span.textContent = optText;
      label.appendChild(span);
      opts.appendChild(label);
    });
    card.appendChild(opts);

    const err = document.createElement('div');
    err.className = 'error-msg';
    err.id = 'err-' + qi;
    err.textContent = 'Selecione uma alternativa antes de corrigir.';
    card.appendChild(err);

    const exp = document.createElement('div');
    exp.className = 'explanation';
    exp.id = 'exp-' + qi;
    exp.textContent = q.explanation;
    card.appendChild(exp);

    container.appendChild(card);
  });
}

function clearError(qi){
  const err = document.getElementById('err-' + qi);
  if(err) err.classList.remove('show');
}

function checkQuiz(){
  let missing = false;
  quizData.forEach((q, qi) => {
    if(userAnswers[qi] === null){
      document.getElementById('err-' + qi).classList.add('show');
      missing = true;
    } else {
      clearError(qi);
    }
  });
  if(missing) return;

  let score = 0;
  quizData.forEach((q, qi) => {
    const chosen = userAnswers[qi];
    q.options.forEach((optText, oi) => {
      const label = document.getElementById('opt-' + qi + '-' + oi);
      label.classList.remove('correct','wrong');
      if(oi === q.correct){
        label.classList.add('correct');
      } else if(oi === chosen){
        label.classList.add('wrong');
      }
    });
    if(chosen === q.correct) score++;
    document.getElementById('exp-' + qi).classList.add('show');
  });

  const box = document.getElementById('score-box');
  box.textContent = 'Você acertou ' + score + ' de ' + quizData.length + ' questões!';
  box.classList.add('show');
  box.scrollIntoView({behavior:'smooth', block:'start'});
  corrected = true;
}

function resetQuiz(){
  userAnswers.fill(null);
  corrected = false;
  document.getElementById('score-box').classList.remove('show');
  renderQuiz();
}

/* ---------------- FLASHCARDS LOGIC ---------------- */
let flashIndex = 0;
let flashOrder = flashData.map((_, i) => i);

function renderFlashcard(){
  const idx = flashOrder[flashIndex];
  const item = flashData[idx];
  document.getElementById('front-tag').textContent = item.tag;
  document.getElementById('front-content').textContent = item.front;
  document.getElementById('back-tag').textContent = 'Resposta';
  document.getElementById('back-content').innerHTML = item.back.replace(/\n/g, '<br>');
  document.getElementById('flashcard').classList.remove('flipped');
  document.getElementById('flash-counter').textContent = 'Card ' + (flashIndex+1) + ' de ' + flashData.length;
}

function flipCard(){
  document.getElementById('flashcard').classList.toggle('flipped');
}

function nextCard(){
  flashIndex = (flashIndex + 1) % flashData.length;
  renderFlashcard();
}

function prevCard(){
  flashIndex = (flashIndex - 1 + flashData.length) % flashData.length;
  renderFlashcard();
}

function shuffleCards(){
  for(let i = flashOrder.length - 1; i > 0; i--){
    const j = Math.floor(Math.random() * (i+1));
    [flashOrder[i], flashOrder[j]] = [flashOrder[j], flashOrder[i]];
  }
  flashIndex = 0;
  renderFlashcard();
}
